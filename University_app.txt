package com.example.uniconnect;

import android.os.Bundle;
import android.view.Menu;
import android.view.MenuInflater;
import android.view.MenuItem;
import android.view.View;
import android.widget.Button;
import android.widget.PopupMenu;
import android.widget.Toast;

import androidx.appcompat.app.AlertDialog;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    Button btnHome, btnAbout, btnAdmission, btnDepartments, btnExit;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // Initializing Buttons
        btnHome = findViewById(R.id.btnHome);
        btnAbout = findViewById(R.id.btnAbout);
        btnAdmission = findViewById(R.id.btnAdmission);
        btnDepartments = findViewById(R.id.btnDepartments);
        btnExit = findViewById(R.id.btnExit);

        // Setting Click Listeners
        btnHome.setOnClickListener(v ->
                Toast.makeText(MainActivity.this, "Welcome to UniConnect!", Toast.LENGTH_SHORT).show()
        );

        btnAbout.setOnClickListener(v -> showAboutUniversityDialog());

        btnAdmission.setOnClickListener(v -> showAdmissionHelpDialog());

        btnDepartments.setOnClickListener(v -> showDepartmentsMenu(v));

        btnExit.setOnClickListener(v -> finish());
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        MenuInflater inflater = getMenuInflater();
        inflater.inflate(R.menu.main_menu, menu);
        return true;
    }

    @Override
    public boolean onOptionsItemSelected(MenuItem item) {
        int itemId = item.getItemId();

        if (itemId == R.id.home) {
            Toast.makeText(this, "Welcome to UniConnect!", Toast.LENGTH_SHORT).show();
            return true;
        } else if (itemId == R.id.about_university) {
            showAboutUniversityDialog();
            return true;
        } else if (itemId == R.id.admission_help) {
            showAdmissionHelpDialog();
            return true;
        } else if (itemId == R.id.exit_app) {
            finish();
            return true;
        }

        return super.onOptionsItemSelected(item);
    }

    // Popup Menu: Department Selection
    private void showDepartmentsMenu(View view) {
        PopupMenu popupMenu = new PopupMenu(this, view);
        popupMenu.getMenuInflater().inflate(R.menu.departments_menu, popupMenu.getMenu());

        popupMenu.setOnMenuItemClickListener(item -> {
            String deptName = item.getTitle().toString();
            Toast.makeText(MainActivity.this, deptName + " Department Selected", Toast.LENGTH_SHORT).show();
            return true;
        });

        popupMenu.show();
    }

    // Dialog for About University
    private void showAboutUniversityDialog() {
        AlertDialog.Builder builder = new AlertDialog.Builder(this);
        builder.setTitle("About UniConnect")
                .setMessage("UniConnect is your official university guide to explore departments, get admission help, and more.")
                .setPositiveButton("OK", null)
                .show();
    }

    // Dialog for Admission Help
    private void showAdmissionHelpDialog() {
        AlertDialog.Builder builder = new AlertDialog.Builder(this);
        builder.setTitle("Admission Support")
                .setMessage("For admission assistance, contact: +91-9876543210\nEmail: help@uniconnect.edu")
                .setPositiveButton("Got It", null)
                .show();
    }
}
