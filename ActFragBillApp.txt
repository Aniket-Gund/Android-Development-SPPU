package com.example.billpaymentapp;
import android.content.Intent;
import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Spinner;

import androidx.fragment.app.Fragment;

public class InputFragment extends Fragment {

    EditText etName, etAmount;
    Spinner spinnerBillType;
    Button btnSubmit;

    // Sample bill types for the Spinner
    String[] billTypes = {"Electricity", "Water", "Internet", "Mobile Recharge", "Gas Bill", "Credit Card"};

    @Override
    public View onCreateView(LayoutInflater inflater, ViewGroup container, Bundle savedInstanceState) {
        // Inflate the fragment layout
        View view = inflater.inflate(R.layout.fragment_input, container, false);

        // Initialize views
        etName = view.findViewById(R.id.etName);
        etAmount = view.findViewById(R.id.etAmount);
        spinnerBillType = view.findViewById(R.id.spinnerBillType);
        btnSubmit = view.findViewById(R.id.btnSubmit);

        // Set up the Spinner with the bill types
        ArrayAdapter<String> adapter = new ArrayAdapter<>(getContext(), android.R.layout.simple_spinner_dropdown_item, billTypes);
        spinnerBillType.setAdapter(adapter);

        // Button click listener
        btnSubmit.setOnClickListener(v -> {
            // Get data from input fields
            String name = etName.getText().toString().trim();
            String amount = etAmount.getText().toString().trim();
            String billType = spinnerBillType.getSelectedItem().toString();

            // Pass data to DisplayActivity
            Intent intent = new Intent(getActivity(), DisplayActivity.class);
            intent.putExtra("name", name);
            intent.putExtra("amount", amount);
            intent.putExtra("billType", billType);
            startActivity(intent);
        });

        return view;
    }
}

package com.example.billpaymentapp;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;

import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;

public class DisplayFragment extends Fragment {

    TextView tvName, tvBillType, tvAmount;

    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container,
                             @Nullable Bundle savedInstanceState) {
        View view = inflater.inflate(R.layout.fragment_display, container, false);

        tvName = view.findViewById(R.id.tvName);
        tvBillType = view.findViewById(R.id.tvBillType);
        tvAmount = view.findViewById(R.id.tvAmount);

        Bundle bundle = getArguments();
        if (bundle != null) {
            tvName.setText("Name: " + bundle.getString("name"));
            tvBillType.setText("Bill Type: " + bundle.getString("billType"));
            tvAmount.setText("Amount: ₹" + bundle.getString("amount"));
        }

        return view;
    }
}
