package com.example.collegeadmissionapp;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;
import android.widget.LinearLayout;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    String[] labels = {
            "Apply Now",
            "Courses Offered",
            "Fee Structure",
            "Campus Tour",
            "Contact Us",
            "About College"
    };

    int[] images = {
            R.drawable.apply,
            R.drawable.courses,
            R.drawable.fee,
            R.drawable.tour,
            R.drawable.contact,
            R.drawable.about
    };

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        int[] layoutIds = {
                R.id.item1, R.id.item2, R.id.item3,
                R.id.item4, R.id.item5, R.id.item6
        };

        for (int i = 0; i < layoutIds.length; i++) {
            final int index = i;
            LinearLayout itemLayout = findViewById(layoutIds[i]);
            itemLayout.setOnClickListener(new View.OnClickListener() {
                @Override
                public void onClick(View view) {
                    Intent intent = new Intent(MainActivity.this, DetailActivity.class);
                    intent.putExtra("label", labels[index]);
                    intent.putExtra("image", images[index]);
                    intent.putExtra("description", getDescription(labels[index]));
                    startActivity(intent);
                }
            });
        }
    }

    private String getDescription(String label) {
        switch (label) {
            case "Apply Now":
                return "Apply for admission through our online portal.";
            case "Courses Offered":
                return "We offer Engineering, Business, Arts, and Science programs.";
            case "Fee Structure":
                return "Affordable fee structure with scholarship opportunities.";
            case "Campus Tour":
                return "Explore our green, tech-enabled, modern campus.";
            case "Contact Us":
                return "Reach us at contact@college.edu or call 1234567890.";
            case "About College":
                return "Our college is dedicated to excellence in education.";
            default:
                return "";
        }
    }
}

package com.example.collegeadmissionapp;

import android.os.Bundle;
import android.widget.ImageView;
import android.widget.TextView;

import androidx.appcompat.app.AppCompatActivity;

public class DetailActivity extends AppCompatActivity {

    ImageView detailImage;
    TextView detailTitle, detailDescription;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_detail);

        detailImage = findViewById(R.id.detailImage);
        detailTitle = findViewById(R.id.detailTitle);
        detailDescription = findViewById(R.id.detailDescription);

        int image = getIntent().getIntExtra("image", 0);
        String title = getIntent().getStringExtra("label");
        String description = getIntent().getStringExtra("description");

        detailImage.setImageResource(image);
        detailTitle.setText(title);
        detailDescription.setText(description);
    }
}
